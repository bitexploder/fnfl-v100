# FNFL: our engine (fnfl/STANDARDS.md). The server reaches it through fn/fnfl.h only; borrowed llama.cpp and ggml CPU
# code is reached through fn/borrowed/ only, no fn/ target links ggml-cuda, and fn/tools/check-includes.py fails the
# build otherwise.

enable_language(CUDA)
find_package(CUDAToolkit REQUIRED)

set(FNFL_WARNINGS -Wall -Wextra)

function(fnfl_target_flags target)
    target_compile_features(${target} PRIVATE cxx_std_17)
    target_include_directories(${target} PUBLIC ${PROJECT_SOURCE_DIR})
    target_compile_options(${target} PRIVATE
        "$<$<COMPILE_LANGUAGE:CXX>:${FNFL_WARNINGS}>"
        "$<$<COMPILE_LANGUAGE:CUDA>:-Xcompiler=-Wall,-Wextra>")
    if (FNFL_WERROR)
        target_compile_options(${target} PRIVATE
            "$<$<COMPILE_LANGUAGE:CXX>:-Werror>"
            "$<$<COMPILE_LANGUAGE:CUDA>:-Werror=all-warnings;-Xcompiler=-Werror>")
    endif()
    set_target_properties(${target} PROPERTIES POSITION_INDEPENDENT_CODE ON)
endfunction()
